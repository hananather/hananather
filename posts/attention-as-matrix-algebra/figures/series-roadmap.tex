% \method{y}{name}{source}{memory or update rule}{what it adds}
\newcommand{\method}[5]{%
  \node[method] at (\xm,#1) {{\bfseries #2}\\[-1pt]{\color{muted}#3}};
  \node[rulecell] at (\xr,#1) {#4};
  \node[adds] at (\xa,#1) {#5};}

% Column titles.
\node[word, anchor=west] at (\xm,0) {method};
\node[word, anchor=west] at (\xr,0) {memory at step $t$, and how step $t$ changes it};
\node[word, anchor=west] at (\xa,0) {what the change adds};

% In this post.
\draw[frame] (0,-0.3) rectangle (\xend,-2.9);
\node[partchip] at (0.8,-1.55) {this post};
\method{-0.75}{Softmax attention}{Vaswani et al. (2017)}{the cache $K_{1:t}$, $V_{1:t}$: \ $2td$ numbers}%
  {every past token stays available; the memory grows with $t$}
\method{-1.6}{Linear attention}{Katharopoulos et al. (2020)}{$S_t = S_{t-1} + \hl{\phi(k_t)\, v_t^\top}$}%
  {a fixed $r \times d$ memory; pairs are only added, nothing leaves}
\method{-2.45}{Delta rule (DeltaNet)}{Schlag et al. (2021); Yang et al. (2024b)}%
  {$S_t = \hl{(I - \beta_t k_t k_t^\top)}\, S_{t-1} + \beta_t k_t v_t^\top$}%
  {erases what $k_t$ reads before it writes $v_t$; trains in chunks}

% Next: decay and gating.
\draw[planned] (0,-3.1) rectangle (\xend,-5.65);
\node[partchip, dash pattern=on 2pt off 1.5pt, fill=white] at (0.8,-4.375) {next};
\method{-3.525}{Decay (Mamba-2)}{Dao and Gu (2024)}{$S_t = \hl{\alpha_t}\, S_{t-1} + k_t v_t^\top$}%
  {forgets every pair at one rate, $\alpha_t \in [0, 1]$}
\method{-4.375}{Gated DeltaNet}{Yang, Kautz and Hatamizadeh (2025)}%
  {$S_t = \hl{\alpha_t}\, (I - \beta_t k_t k_t^\top)\, S_{t-1} + \beta_t k_t v_t^\top$}%
  {both rows above: fast erasure and targeted edits}
\method{-5.225}{Kimi Delta Attention (KDA)}{Kimi Team (2025)}%
  {$S_t = (I - \beta_t k_t k_t^\top)\, \hl{\operatorname{Diag}(\alpha_t)}\, S_{t-1} + \beta_t k_t v_t^\top$}%
  {one forget rate for each feature channel}

% Then: hybrids.
\draw[planned] (0,-5.85) rectangle (\xend,-6.7);
\node[partchip, dash pattern=on 2pt off 1.5pt, fill=white] at (0.8,-6.275) {then};
\method{-6.275}{Hybrid (Kimi Linear)}{Kimi Team (2025)}{three KDA layers for each full-attention layer}%
  {the full-attention layers see every past token; KV cache up to 75\% smaller}

\node[word, anchor=north west, align=left] at (0,-6.9) {%
  Linear attention reads $o_t = S_t^\top \phi(q_t) / z_t^\top \phi(q_t)$. From the delta rule on, $\phi$ is folded into $q_t$ and $k_t$, and the recurrent rows read $o_t = S_t^\top q_t$.\\
  The highlighted term is the main change in each row. In the delta-rule rows, keys have unit length and $\beta_t \in [0, 1]$ is a write strength; the decay row folds its input scale into $k_t$.};
