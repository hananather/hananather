% Generating one token at a time, for one head. Top: softmax attention keeps the keys and
% values of every past token (the KV cache), so step t reads 2t vectors. Bottom: causal linear
% attention keeps one r x d matrix S_t (Katharopoulos et al., 2020), written by a
% rank-one outer product and read with the query; the normalizer z_t is left out of the drawing.
% Products follow the matrix-product diagram: A to the left of C, B above it. d = r = 3 are
% drawn; the newest row of each cache is darker. The numbers at the bottom come from
% attention_numbers.py. Words and math 7 pt.
\documentclass[tikz,border=2pt]{standalone}
\input{attention-style}
\input{memory-numbers}

\begin{document}
\begin{tikzpicture}
\def\dd{3}
\def\step{2.9}                           % distance between two steps
\def\first{3.4}                          % x of step 1
\tikzset{plus/.style={circle, draw=ink, line width=0.8pt, fill=white, inner sep=0pt,
  minimum size=3.2mm, path picture={\draw[ink, line width=0.8pt]
  ([xshift=-0.9mm]path picture bounding box.center) -- ([xshift=0.9mm]path picture bounding box.center)
  ([yshift=-0.9mm]path picture bounding box.center) -- ([yshift=0.9mm]path picture bounding box.center);}}}

% ---- Top: softmax attention reads a cache that grows. ----
\foreach \t in {1,...,4} {
  \pgfmathsetmacro\x{\first + (\t - 1)*\step}
  \node[title, anchor=south] at (\x,-0.25) {$t = \t$};
  \grid{q\t}{\x - 1.2}{-0.55}{1}{\dd}{query}{}
  \grid{K\t}{\x - 1.2}{-1.4}{\t}{\dd}{key}{\row{keyline!60}{\t}{\dd}}
  \grid{V\t}{\x + 0.18}{-1.4}{\t}{\dd}{value}{\row{valueline!55}{\t}{\dd}}
  \grid{o\t}{\x + 0.18}{-3.15}{1}{\dd}{output}{}
  \draw[branch] (q\t.south) -- (K\t.north);
  \draw[branch] ([yshift=-0.17cm]K\t.north east) -- ([yshift=-0.17cm]V\t.north west);
  \draw[branch] (V\t.south) -- (o\t.north);
}
\node[sym, anchor=east] at ([xshift=-4pt]q1.west) {$q_t^\top$};
\node[matname, anchor=east, align=right] at ([xshift=-4pt]K1.west) {$K_{1:t}$, $V_{1:t}$\\[-1pt]{\color{muted}the cache: $t$ rows each}};
\node[sym, anchor=east] at ([xshift=-4pt]o1.west -| K1.west) {$o_t^\top$};

\node[anchor=north west, align=flush left, text width=5.0cm] at (13.75,-0.3) {%
  {\bfseries Softmax attention keeps every}\\{\bfseries key and value}\\[3pt]
  $o_t = \displaystyle\sum_{j=1}^{t} \frac{e^{q_t^\top k_j / \sqrt{d}}}{\sum_{l=1}^{t} e^{q_t^\top k_l / \sqrt{d}}}\, v_j$\\[3pt]
  {\color{muted}step $t$ reads $2t$ vectors,}\\{\color{muted}about $4td$ operations}\\[1pt]
  {\color{muted}cache: $2td$ numbers, growing with $t$}};

\draw[rule] (0,-4.05) -- (18.6,-4.05);

% ---- Bottom: linear attention writes and reads one matrix of fixed size. ----
\def\vrow{-4.6}  \def\urow{-5.06}  \def\srow{-6.5}  \def\orow{-7.64}
\grid{S0}{0.5 - 0.51}{\srow}{\dd}{\dd}{state}{}
\node[sym, anchor=south] at (S0.north) {$S_0 = 0$};
\foreach \t [count=\p from 0] in {1,...,4} {
  \pgfmathsetmacro\x{\first + (\t - 1)*\step}
  \pgfmathsetmacro\m{\x - 1.45}          % x of the write and of its plus sign
  % Write: phi(k_t) to the left of the outer product, v_t^T above it.
  \grid{v\t}{\m - 0.51}{\vrow}{1}{\dd}{value}{}
  \grid{k\t}{\m - 0.51 - 0.12 - \cell}{\urow}{\dd}{1}{key}{}
  \grid{U\t}{\m - 0.51}{\urow}{\dd}{\dd}{state}{}
  % The state, and the plus sign on the arrow into it.
  \grid{S\t}{\x - 0.51}{\srow}{\dd}{\dd}{state}{\fill[stateline!30] (0,0) rectangle (\dd,-\dd);}
  \node[sym, anchor=south] at (S\t.north) {$S_\t$};
  \node[plus] (plus\t) at (\m, {\srow - 1.5*\cell}) {};
  \draw[route] (S\p.east) -- (plus\t.west);
  \draw[flow] (plus\t.east) -- (S\t.west);
  \draw[flow] (U\t.south) -- (plus\t.north);
  % Read: phi(q_t)^T to the left of o_t^T, S_t above it.
  \grid{f\t}{\x - 0.51 - 0.12 - 1.02}{\orow}{1}{\dd}{query}{}
  \grid{r\t}{\x - 0.51}{\orow}{1}{\dd}{output}{}
}
\node[sym, anchor=east] at ([xshift=-3pt]v1.west) {$v_t^\top$};
\node[sym, anchor=east] at ([xshift=-3pt]k1.west) {$\phi(k_t)$};
\node[sym, anchor=east] at ([xshift=-3pt]f1.west) {$\phi(q_t)^\top$};
\node[sym, anchor=west] at ([xshift=3pt]r4.east) {$o_t^\top$};

\node[anchor=north west, align=flush left, text width=5.0cm] at (13.75,-4.5) {%
  {\bfseries Linear attention keeps one $r \times d$ matrix}\\[3pt]
  $S_t = S_{t-1} + \phi(k_t)\, v_t^\top$\\[1pt]
  $z_t = z_{t-1} + \phi(k_t)$\\[1pt]
  $o_t = S_t^\top \phi(q_t) \,\big/\, z_t^\top \phi(q_t)$\\[3pt]
  {\color{muted}each write is a rank-one $r \times d$ matrix}\\[1pt]
  {\color{muted}step $t$: about $4rd$ operations}\\[1pt]
  {\color{muted}state: $rd + r$ numbers, the same for every $t$}\\[1pt]
  {\color{muted}the drawing leaves out $z_t$ and the division}};

% ---- What the two memories cost at a real size. ----
\node[keybox, anchor=north, align=center] at (9.3,-8.55) {%
  {\bfseries One head with $d = r = \HeadDim$, after $t = \Tokens$ tokens:} the cache holds
  $\CacheNumbers$ numbers and the state $\StateNumbers$, $\CacheRatio$ times fewer.\\
  The state is the size of the cache after about $\BreakEven$ tokens.};
