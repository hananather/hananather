% Training the delta rule in chunks (Yang et al., 2024b, Section 3 and eq. 10). Inside a chunk
% of C tokens that starts from the state S, the pseudo-values U solve one unit lower-triangular
% system. Then the chunk is chunked linear attention with U in place of V. The product of the
% chunk's C transitions has the compact WY form I - K^T T K (Schreiber and Van Loan, 1989), with
% T from the UT transform (Joffrain et al., 2006). C = 6 and d = 3 are drawn. Words and math 7 pt.

\node[subtitle, anchor=west] at (0,-0.42) {one $C \times C$ system, unit lower triangular, solved row by row};

\grid{IL}{0.2}{\toprow}{\C}{\C}{neutral}{%

  \foreach \i in {1,...,\C} { \entry{white}{\i}{\i} }}

\grid{U}{2.9}{\toprow}{\C}{\dd}{pseudo}{}
\node[sym] at ($(U.west)!0.5!(IL.east)$) {$\cdot$};
\grid{B}{4.55}{\toprow}{\C}{\dd}{value}{}

  {\color{muted}matrix of ones}};

% ---- Why it is exact: the product of the transitions. ----
\node[keybox, anchor=north, align=center] at (9.3,-4.4) {%
  {\bfseries Why the chunk is exact.} The chunk multiplies the state by its $C$ transitions, and their product has a compact form:\\[3pt]
  $(I - \beta_C k_C k_C^\top) \cdots (I - \beta_1 k_1 k_1^\top) = I - K^\top T K$, \qquad $T = (I + L)^{-1} \operatorname{diag}(\beta)$, \qquad $U = T\, (V - K S)$.\\[3pt]
  {\color{muted}This is the compact WY representation of a product of generalized Householder matrices (Schreiber and Van Loan, 1989), with $T$ from the UT transform (Joffrain et al., 2006).}\\
  {\color{muted}No state inside the chunk is ever formed. Yang et al. (2024b) use this form to train DeltaNet in parallel; their equation (10) gives $T$.}};
