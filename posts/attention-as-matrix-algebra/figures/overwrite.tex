% A new value under an old key, written by the additive rule of linear attention and by the
% delta rule (Schlag, Irie and Schmidhuber, 2021). Both start from S_2 of the interference
% figure: k_1 = (1, 0) holds v_1 = (3, 1) and k_2 = (0, 1) holds v_2 = (-1, 2). The third write
% is (k_3, v_3) with k_3 = k_1 and v_3 = (0, 4). Keys amber, values green, the pseudo-value
% deeper green, the state lavender, reads slate, the stale value orange. The notebook checks
% every number. Words and math 7 pt.
\documentclass[tikz,border=2pt]{standalone}
\input{attention-style}

\begin{document}
\begin{tikzpicture}
\tikzset{op/.style={inner sep=1pt}, gap/.style={right=1.2mm of #1}}

\node[anchor=west, align=left] at (0,0) {%
  {\bfseries Write a new value under an old key.} Start from $S_2$ of the interference figure, where $k_1 = (1, 0)$ holds $v_1 = (3, 1)$
  and $k_2 = (0, 1)$ holds $v_2 = (-1, 2)$.\\
  The third write is $(k_3, v_3)$ with $k_3 = k_1$ and $v_3 = (0, 4)$.};

% ---- The additive rule. ----
\node[title, anchor=west] at (0,-0.95) {Additive rule (linear attention)};
\node[op, anchor=west] (a0) at (0,-1.75) {$S_3 = S_2 + k_3 v_3^\top =$};
\matrix[entries=state, gap=a0] (aS2) {3 & 1 \\ -1 & 2 \\};
\node[op, gap=aS2] (a1) {$+$};
\matrix[entries=key, gap=a1] (ak) {1 \\ 0 \\};
\matrix[entries=value, gap=ak] (av) {0 & 4 \\};
\node[op, gap=av] (a2) {$=$};
\matrix[entries=state, gap=a2] (aS3) {3 & 5 \\ -1 & 2 \\};

\node[op, anchor=west] (ar1) at (9.6,-1.5) {$k_1^\top S_3 =$};
\matrix[entries=output, gap=ar1] (ao1) {3 & 5 \\};
\node[op, gap=ao1] (ap1) {$= v_3^\top +$};
\matrix[entries=crosstalk, gap=ap1] (ax1) {3 & 1 \\};
\node[word, anchor=west, gap=ax1] {the old value $v_1$ stays};
\node[op, anchor=west] (ar2) at (9.6,-2.2) {$k_2^\top S_3 =$};
\matrix[entries=output, gap=ar2] (ao2) {-1 & 2 \\};
\node[op, gap=ao2] {$= v_2^\top$ \ \checkmark};

\draw[rule] (0,-2.75) -- (18.2,-2.75);

% ---- The delta rule: read, correct, write. ----
\node[title, anchor=west] at (0,-3.2) {Delta rule, with $\beta_3 = 1$};

\node[title, anchor=west] at (0,-3.75) {1. Read};
\node[op, anchor=west] (d0) at (0,-4.45) {$k_3^\top S_2 =$};
\matrix[entries=output, gap=d0] (dold) {3 & 1 \\};
\node[word, anchor=north west] at (0,-4.9) {what $k_3$ reads now};

\node[title, anchor=west] at (4.2,-3.75) {2. Correct};
\node[op, anchor=west] (c0) at (4.2,-4.45) {$u_3^\top = v_3^\top - k_3^\top S_2 =$};
\matrix[entries=value, gap=c0] (cv) {0 & 4 \\};
\node[op, gap=cv] (c1) {$-$};
\matrix[entries=output, gap=c1] (co) {3 & 1 \\};
\node[op, gap=co] (c2) {$=$};
\matrix[entries=pseudo, gap=c2] (cu) {-3 & 3 \\};
\node[word, anchor=north west] at (4.2,-4.9) {only the difference is written: the pseudo-value $u_3$};

\node[title, anchor=west] at (0,-5.55) {3. Write};
\node[op, anchor=west] (w0) at (0,-6.35) {$S_3 = S_2 + k_3 u_3^\top =$};
\matrix[entries=state, gap=w0] (wS2) {3 & 1 \\ -1 & 2 \\};
\node[op, gap=wS2] (w1) {$+$};
\matrix[entries=key, gap=w1] (wk) {1 \\ 0 \\};
\matrix[entries=pseudo, gap=wk] (wu) {-3 & 3 \\};
\node[op, gap=wu] (w2) {$=$};
\matrix[entries=state, gap=w2] (wS3) {0 & 4 \\ -1 & 2 \\};

\node[op, anchor=west] (dr1) at (9.6,-6.0) {$k_1^\top S_3 =$};
\matrix[entries=output, gap=dr1] (do1) {0 & 4 \\};
\node[op, gap=do1] {$= v_3^\top$ \ \checkmark};
\node[op, anchor=west] (dr2) at (9.6,-6.7) {$k_2^\top S_3 =$};
\matrix[entries=output, gap=dr2] (do2) {-1 & 2 \\};
\node[op, gap=do2] {$= v_2^\top$ \ \checkmark};

% ---- The identity behind the third step. ----
\node[keybox, anchor=north west, align=left] at (13.4,-5.55) {%
  With a key of unit length, the write leaves\\[2pt]
  $S_t^\top k_t = (1 - \beta_t)\, S_{t-1}^\top k_t + \beta_t\, v_t$,\\[2pt]
  {\color{muted}a mix of the old read and the new value:}\\
  {\color{muted}$\beta_t = 1$ replaces it, $\beta_t = 0$ keeps it.}};
\end{tikzpicture}
\end{document}
