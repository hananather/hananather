% A worked example of the fixed-size state: three key-value pairs written into a 2 x 2
% memory. For small numbers, phi(k) = k, and a read divides k^T S by |k|^2 in place of the
% normalizer z_t. Keys are amber, values green, the state lavender,
% reads slate and the crosstalk orange. attention_numbers.py checks every number (check 6).
% The capacity argument is Schlag, Irie and Schmidhuber (2021). Words and math 7 pt.
\documentclass[tikz,border=2pt]{standalone}
\input{attention-style}

\begin{document}
\begin{tikzpicture}
\tikzset{op/.style={inner sep=1pt}, gap/.style={right=1.2mm of #1}}
\def\eqy{-1.05}

% ---- The three writes. ----
\node[title, anchor=west] at (0,0) {1. Write $(k_1, v_1)$};
\node[op, anchor=west] (a) at (0,\eqy) {$S_1 =$};
\matrix[entries=key, gap=a] (k1) {1 \\ 0 \\};
\matrix[entries=value, gap=k1] (v1) {3 & 1 \\};
\node[op, gap=v1] (b) {$=$};
\matrix[entries=state, gap=b] (S1) {3 & 1 \\ 0 & 0 \\};

\node[title, anchor=west] at (5.0,0) {2. Write $(k_2, v_2)$};
\node[op, anchor=west] (c) at (5.0,\eqy) {$S_2 = S_1 \,+$};
\matrix[entries=key, gap=c] (k2) {0 \\ 1 \\};
\matrix[entries=value, gap=k2] (v2) {-1 & 2 \\};
\node[op, gap=v2] (d) {$=$};
\matrix[entries=state, gap=d] (S2) {3 & 1 \\ -1 & 2 \\};

\node[title, anchor=west] at (10.6,0) {3. Write $(k_3, v_3)$};
\node[op, anchor=west] (e) at (10.6,\eqy) {$S_3 = S_2 \,+$};
\matrix[entries=key, gap=e] (k3) {1 \\ 1 \\};
\matrix[entries=value, gap=k3] (v3) {2 & 0 \\};
\node[op, gap=v3] (f) {$=$};
\matrix[entries=state, gap=f] (S3) {5 & 1 \\ 1 & 2 \\};

% ---- Reads: a key times the state. ----
\def\ra{-2.45} \def\rb{-3.15} \def\rc{-3.85}
\node[word, anchor=west] at (0,-1.95) {a write is the outer product $k_t v_t^\top$};
\node[op, anchor=west] (r1) at (0,\ra) {$k_1^\top S_1 =$};
\matrix[entries=output, gap=r1] (o1) {3 & 1 \\};
\node[op, gap=o1] {$= v_1^\top$ \ \checkmark};
\node[word, anchor=west, align=left] at (0,-4.45) {the state holds $2 \times 2 = 4$ numbers\\after every write};

\node[op, anchor=west] (r2) at (5.0,\ra) {$k_1^\top S_2 =$};
\matrix[entries=output, gap=r2] (o2) {3 & 1 \\};
\node[op, gap=o2] {$= v_1^\top$ \ \checkmark};
\node[op, anchor=west] (r3) at (5.0,\rb) {$k_2^\top S_2 =$};
\matrix[entries=output, gap=r3] (o3) {-1 & 2 \\};
\node[op, gap=o3] {$= v_2^\top$ \ \checkmark};
\node[word, anchor=west, align=left] at (5.0,-4.45) {$k_1 \perp k_2$: each read returns\\its value exactly};

\node[op, anchor=west] (r4) at (10.6,\ra) {$k_1^\top S_3 =$};
\matrix[entries=output, gap=r4] (o4) {5 & 1 \\};
\node[op, gap=o4] (p4) {$= v_1^\top +$};
\matrix[entries=crosstalk, gap=p4] {2 & 0 \\};
\node[op, anchor=west] (r5) at (10.6,\rb) {$k_2^\top S_3 =$};
\matrix[entries=output, gap=r5] (o5) {1 & 2 \\};
\node[op, gap=o5] (p5) {$= v_2^\top +$};
\matrix[entries=crosstalk, gap=p5] {2 & 0 \\};
\node[op, anchor=west] (r6) at (10.6,\rc) {$k_3^\top S_3 / \lVert k_3 \rVert^2 =$};
\matrix[entries=output, gap=r6] (o6) {3 & 1.5 \\};
\node[op, gap=o6] (p6) {$= v_3^\top +$};
\matrix[entries=crosstalk, gap=p6] {1 & 1.5 \\};
\node[word, anchor=west, align=left] at (10.6,-4.45) {this third key is orthogonal to neither:\\every read now carries crosstalk};

% ---- The identity behind every read. ----
\node[anchor=north] at (8.4,-5.15) {%
  $k_i^\top S_t \;=\; \underbrace{\lVert k_i \rVert^2\, v_i^\top}_{\text{signal}} \;+\;
   \underbrace{\textstyle\sum_{j \le t,\, j \ne i} (k_i^\top k_j)\, v_j^\top}_{\text{crosstalk}}$};
\node[word, anchor=north, align=center] at (8.4,-6.05) {Every read is free of crosstalk, for any values, only if the keys are orthogonal, and $\mathbb{R}^r$ holds at most $r$ of them (Schlag, Irie and Schmidhuber, 2021).\\
  Here $r = d = 2$ and $\phi(k) = k$. In place of the normalizer $z_t$, a read divides by $\lVert k_i \rVert^2$, which is 1 for $k_1$ and $k_2$.};
\end{tikzpicture}
